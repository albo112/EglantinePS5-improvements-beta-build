namespace Common {
template<class T> class ScratchBuffer {
  std::vector<T> buffer;
public:
  using value_type=T; using element_type=T;
  void resize_destructive(size_t n){++input_requests;max_input=std::max(n,max_input);if(fail_input||n>1024*1024)throw std::bad_alloc();buffer.resize(n);}
  T* data(){return buffer.data();} const T* data()const{return buffer.data();}
  size_t size()const{return buffer.size();} size_t capacity()const{return buffer.capacity();}
  T* begin(){return data();} T* end(){return data()+size();}
  T& operator[](size_t n){return buffer.at(n);}
};
